def calculate_true_net_profit(declared_mrr, stripe_tx_count, api_query_units, fixed_server_cost): # 扣除 Stripe 基础手续费率: 2.9% + 0.30 美元/笔 stripe_fee = (declared_mrr * 0.029) + (stripe_tx_count * 0.30) # 扣除服务 API 真实调用成本 (百万次调用阶梯计费) api_cost = (api_query_units / 1_000_000) * 2.50 # 真实扣税前净利润与净利率 true_net_profit = declared_mrr - stripe_fee - api_cost - fixed_server_cost net_margin = (true_net_profit / declared_mrr) if declared_mrr > 0 else 0.0 return { "true_net_profit": round(true_net_profit, 2), "true_net_margin": f"{round(net_margin * 100, 1)}%", "water_deducted": round(declared_mrr - true_net_profit, 2) }
